set(TELEMETRY_PLOTTING_LIB_NAME "TelemetryPlottingLib")
set(TELEMETRY_PLOTTING_LIB_HEADERS "${TELEMETRY_PLOTTING_LIB_NAME}_HEADERS")
set(TELEMETRY_PLOTTING_LIB_SOURCES "${TELEMETRY_PLOTTING_LIB_NAME}_SOURCES")

file(GLOB_RECURSE TELEMETRY_PLOTTING_LIB_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
file(GLOB_RECURSE TELEMETRY_PLOTTING_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

add_library(${TELEMETRY_PLOTTING_LIB_NAME} ${TELEMETRY_PLOTTING_LIB_SOURCES} ${TELEMETRY_PLOTTING_LIB_HEADERS})
add_library(${PROJECT_NAME}::${TELEMETRY_PLOTTING_LIB_NAME} ALIAS ${TELEMETRY_PLOTTING_LIB_NAME})

set_target_properties(${TELEMETRY_PLOTTING_LIB_NAME} PROPERTIES
        CXX_STANDARD 17
        CXX_STANDARD_REQUIRED YES
        CXX_EXTENSIONS OFF)

include(AddQCustomPlot)
target_link_libraries(${TELEMETRY_PLOTTING_LIB_NAME}
        PUBLIC
                ${PROJECT_NAME}::TelemetryLib
        PRIVATE
                ${PROJECT_NAME}_ProjectSettings
                QCustomPlot::QCustomPlot)

target_include_directories(${TELEMETRY_PLOTTING_LIB_NAME} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
