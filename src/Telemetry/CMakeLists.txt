set(TELEMETRY_LIB_NAME "TelemetryLib")
set(TELEMETRY_LIB_HEADERS "${TELEMETRY_LIB_NAME}_HEADERS")
set(TELEMETRY_LIB_SOURCES "${TELEMETRY_LIB_NAME}_SOURCES")

file(GLOB_RECURSE TELEMETRY_LIB_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
file(GLOB_RECURSE TELEMETRY_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

add_library(${TELEMETRY_LIB_NAME} ${TELEMETRY_LIB_SOURCES} ${TELEMETRY_LIB_HEADERS})
add_library(${PROJECT_NAME}::${TELEMETRY_LIB_NAME} ALIAS ${TELEMETRY_LIB_NAME})

set_target_properties(${TELEMETRY_LIB_NAME} PROPERTIES
        CXX_STANDARD 23
        CXX_STANDARD_REQUIRED YES
        CXX_EXTENSIONS OFF)

target_link_libraries(${TELEMETRY_LIB_NAME} PRIVATE ${PROJECT_NAME}_ProjectSettings)
target_include_directories(${TELEMETRY_LIB_NAME} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
